## 4. 新增表（4 张）

### 4.1 `ExamCountdown`（M1 立即建，倒计时数据源）

```python
class ExamCountdown(Base):
    __tablename__ = "exam_countdown"

    id: Mapped[int] = mapped_column(primary_key=True)
    user_id: Mapped[int] = mapped_column(ForeignKey("app_user.id"), index=True)
    label: Mapped[str] = mapped_column(String(64))           # "2027 国考" / "2027 江苏省考"
    exam_type: Mapped[str] = mapped_column(String(32))       # guokao / shengkao / other
    exam_date: Mapped[date] = mapped_column(Date)            # 考试日期
    is_primary: Mapped[bool] = mapped_column(default=False)  # 多个倒计时中"主"显示
    note: Mapped[Optional[str]] = mapped_column(String(256), nullable=True)
    created_at: Mapped[datetime] = mapped_column(default=now_cst)
    updated_at: Mapped[datetime] = mapped_column(default=now_cst, onupdate=now_cst)

    __table_args__ = (
        Index("idx_exam_countdown_user", "user_id", "exam_date"),
    )
```

**API**：

```
GET    /api/countdown              # 列出用户全部倒计时
POST   /api/countdown              # 新增 {label, exam_type, exam_date, is_primary}
PATCH  /api/countdown/{id}         # 更新（含设为 primary 时清空其他 primary）
DELETE /api/countdown/{id}         # 删除
```

**业务规则**：
- 同一用户最多一个 `is_primary=True`（PATCH/POST 时清空其他）
- 考试日期必须 > 今天（可加软校验）

> **核心：v1 预埋不暴露 UI，v2 社区直接读取，不补采数据**

```python
class ActivityEvent(Base):
    __tablename__ = "activity_event"

    id: Mapped[int] = mapped_column(primary_key=True)
    user_id: Mapped[int] = mapped_column(ForeignKey("app_user.id"), index=True)
    type: Mapped[str] = mapped_column(String(32), index=True)  # 见枚举
    payload_json: Mapped[str] = mapped_column(Text)            # JSON 字符串
    is_public: Mapped[bool] = mapped_column(default=False)     # v2 社区默认私密

    __table_args__ = (
        Index("idx_ae_user_type_time", "user_id", "type", "created_at"),
```

**通用函数**：

```python
# server/app/services/activity_service.py
def record_event(
    db: Session,
    user_id: int,
    type: str,
    payload: dict,
    is_public: bool = False,
) -> ActivityEvent:
    """统一写入活动事件，v1 默认 is_public=False"""
    event = ActivityEvent(
        user_id=user_id,
        type=type,
        payload_json=json.dumps(payload, ensure_ascii=False, default=str),
        is_public=is_public,
    )
    db.add(event)
    db.commit()
    return event
```

**v1 读取点**（不暴露 UI，库可查）：

- 个人足迹页（user/growth.vue）
- 排行榜（user/rank.vue）
- 上岸卡数据源（M4）
- v2 社区流（暂未实现）

**v1 不暴露 UI 的原因**（来自最终方案 §8）：

- 避免"提前做重、偏离备考主线"
- 避免社交类目审核风险（v2 才有）
- 活动事件全部入库，等 v2 社区启动时无需回填

```python
# server/scripts/export_to_zhixing.py
import json, os
from datetime import datetime
from app.database import SessionLocal
from app.models import (
    Article, Question, WrongAnswer, ManualWrong,
    KnowledgeNode, RmrbArticle, ShenlunMineLog, ShenlunNormTerm,
)

EXPORT_DIR = "export"
os.makedirs(EXPORT_DIR, exist_ok=True)

def dump(filename, rows):
    with open(f"{EXPORT_DIR}/{filename}", "w", encoding="utf-8") as f:
        json.dump(
            [{c.name: getattr(r, c.name) for c in r.__table__.columns}
             for r in rows],
            f,
            ensure_ascii=False,
            indent=2,
            default=str,
        )

db = SessionLocal()
try:
    dump("articles.json", db.query(Article).all())
    dump("questions.json", db.query(Question).all())
    # 错题类 user_id 需重置为 1
    wrong_dump = []
    for r in db.query(WrongAnswer).all():
        d = {c.name: getattr(r, c.name) for c in r.__table__.columns}
        d["user_id"] = 1
        wrong_dump.append(d)
    with open(f"{EXPORT_DIR}/wrong_answers.json", "w", encoding="utf-8") as f:
        json.dump(wrong_dump, f, ensure_ascii=False, indent=2, default=str)
    # 同上 manual_wrong + shenlun_mines
    dump("knowledge_nodes.json", db.query(KnowledgeNode).all())
    dump("rmrb_articles.json", db.query(RmrbArticle).all())
    dump("shenlun_terms.json", db.query(ShenlunNormTerm).all())
    print(f"导出完成 -> {EXPORT_DIR}/")
finally:
    db.close()
```

```python
# server/scripts/import_from_zhixing.py
import json, os, sys
from app.database import SessionLocal
from app.models import (
    Article, Question, KnowledgeNode, RmrbArticle,
    ShenlunNormTerm, WrongAnswer, ManualWrong, ShenlunMineLog,
)

EXPORT_DIR = sys.argv[1] if len(sys.argv) > 1 else "../zhengkao-tong/server/export"
TARGET_USER_ID = 1  # 导入到本人账号

def load(name):
    with open(f"{EXPORT_DIR}/{name}", "r", encoding="utf-8") as f:
        return json.load(f)

def import_articles(db):
    rows = load("articles.json")
    for r in rows:
        old_id = r.pop("id", None)
        article = Article(**r)
        db.add(article)
    db.commit()
    print(f"articles: {len(rows)}")

def import_questions(db):
    rows = load("questions.json")
    for r in rows:
        r.pop("id", None)
        db.add(Question(**r))
    db.commit()
    print(f"questions: {len(rows)}")

def import_knowledge(db):
    rows = load("knowledge_nodes.json")
    for r in rows:
        r.pop("id", None)
        # associated_articles/wrongs 字段在迁移后加
        db.add(KnowledgeNode(**r))
    db.commit()
    print(f"knowledge_nodes: {len(rows)}")

# ... rmrb_articles / shenlun_terms / wrong_answers / manual_wrong / shenlun_mines 同上

db = SessionLocal()
try:
    import_articles(db)
    import_questions(db)
    import_knowledge(db)
    # ... 其他表
    print("导入完成")
finally:
    db.close()
```

## 6. 数据库初始化与迁移

### 6.1 启动初始化（保持知行逻辑）

`server/app/main.py` 启动时：

1. `Base.metadata.create_all()` 创建所有表
2. 兼容补列逻辑保留（适用于老 SQLite 升级）
